// Ganti fungsi getRekap yang lama dengan ini + tambah getRekapDetail function getRekap(bulan, tahun, kelas, kode) { // biarkan yang lama tetap ada untuk ringkasan return getRekapDetail(bulan, tahun, kelas, kode); } function getRekapDetail(bulan, tahun, kelas, kode) { const ss = SpreadsheetApp.openById(SS_ID); const shAbsen = ss.getSheetByName(SHEET_ABSENSI); const shSiswa = ss.getSheetByName(SHEET_SISWA); const dataAbsen = shAbsen.getDataRange().getValues(); const dataSiswa = shSiswa.getDataRange().getValues(); const tz = 'Asia/Jakarta'; // siswa let siswaList = []; for (let i = 1; i < dataSiswa.length; i++) { if (dataSiswa[i][0] && (!kelas || dataSiswa[i][2] === kelas)) { siswaList.push({ nisn: String(dataSiswa[i][0]).trim(), nama: dataSiswa[i][1], kelas: dataSiswa[i][2] }); } } // kumpulkan tanggal unik + kehadiran const tanggalSet = new Set(); const hadirMap = {}; // nisn -> Set tanggal for (let i = 1; i < dataAbsen.length; i++) { const nisn = String(dataAbsen[i][0]).trim(); const rowKode = String(dataAbsen[i][1]).trim(); if (kode && rowKode !== String(kode).trim()) continue; const rowMulai = dataAbsen[i][3]; let d = null; if (rowMulai instanceof Date) d = rowMulai; else if (rowMulai) { d = new Date(rowMulai); if (isNaN(d.getTime())) d = null; } if (!d) continue; if ((d.getMonth() + 1) !== parseInt(bulan) || d.getFullYear() !== parseInt(tahun)) continue; const tgl = Utilities.formatDate(d, tz, 'd/M'); tanggalSet.add(tgl); if (!hadirMap[nisn]) hadirMap[nisn] = new Set(); hadirMap[nisn].add(tgl); } // urutkan tanggal const tanggalList = Array.from(tanggalSet).sort((a, b) => { const [da, ma] = a.split('/').map(Number); const [db, mb] = b.split('/').map(Number); return ma === mb ? da - db : ma - mb; }); const result = siswaList.map(s => { const set = hadirMap[s.nisn] || new Set(); const hadirPerTgl = {}; tanggalList.forEach(t => hadirPerTgl[t] = set.has(t) ? 'H' : ''); return { nisn: s.nisn, nama: s.nama, kelas: s.kelas, hadirPerTgl, jml: set.size }; }); result.sort((a, b) => a.nama.localeCompare(b.nama)); const kegMap = {}; getKegiatan().forEach(k => kegMap[k.kode] = k.nama); return { data: result, tanggalList, namaKegiatan: kode ? (kegMap[String(kode).trim()] || kode) : 'Semua Kegiatan', totalPertemuan: tanggalList.length }; }
