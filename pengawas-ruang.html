/** * APPS SCRIPT — PENGAWAS RUANG * ------------------------------------------------------------ * Script ini di-deploy sebagai WEB APP dan berperan sebagai * "jembatan" tunggal antara aplikasi (pengawas-ruang.html) dan * DUA spreadsheet: * * 1) Sheet PENGAWAS (tempat script ini di-bind / container-bound) * https://docs.google.com/spreadsheets/d/13IlK4ep8Db8YssWRg7U5OxIy-VZWEk3KVqmPK72KauA * Kolom Sheet1: A=Timestamp, B=RUANG, C=PENGAWAS * Kolom Sheet2!A:A: daftar nama guru untuk dropdown "Nama Pengawas" * * 2) Sheet SUMBER TOKEN (dibaca DAN DITULIS lewat openById, TIDAK * pernah dikirim/terlihat di kode HTML sisi klien) * https://docs.google.com/spreadsheets/d/1T4xMUrr8orUE5j-X6v1o9zg6Cf7V1VwzupE6ZXtpcbg * - Nilai token diambil dari Sheet1!B6 * - Sheet "Otorisasi" (BARU): dipakai untuk mencatat konfirmasi * manual pengawas terhadap siswa yang kamera HP-nya gagal scan * QR Token di aplikasi Asesmen. asesmen.html membaca sheet ini * langsung lewat gviz (sama seperti cara baca token B6) untuk * mengecek apakah siswa sudah dikonfirmasi. * * CARA PASANG: * 1. Buka spreadsheet PENGAWAS (#1) di atas. * 2. Extensions > Apps Script. * 3. Hapus isi default, tempel seluruh isi file ini. * 4. Klik Deploy > New deployment. * - Type: Web app * - Execute as: Me * - Who has access: Anyone * 5. Salin URL Web App yang muncul, tempelkan ke variabel * CONFIG.APPS_SCRIPT_URL pada file pengawas-ruang.html. * 6. Pastikan akun yang men-deploy memiliki akses ke spreadsheet * SUMBER TOKEN (#2): * - Minimal "Viewer" untuk baca token (fitur lama) * - WAJIB "Editor" untuk fitur konfirmasi manual (fitur baru) * 7. Di spreadsheet SUMBER TOKEN (#2), buat 1 sheet baru bernama * persis "Otorisasi" (tanpa perlu isi header — script akan * membuatnya otomatis kalau belum ada saat konfirmasi pertama). * ------------------------------------------------------------ */ var SHEET_TOKEN_ID = '1T4xMUrr8orUE5j-X6v1o9zg6Cf7V1VwzupE6ZXtpcbg'; var SHEET_TOKEN_NAME = 'Sheet1'; var SHEET_TOKEN_CELL = 'B6'; var SHEET_OTORISASI_NAME = 'Otorisasi'; // sheet baru di spreadsheet SUMBER TOKEN (#2) var SHEET_PENGAWAS_NAME = 'Sheet1'; var SHEET_GURU_NAME = 'Sheet2'; var SHEET_GURU_COLUMN = 'A'; function doPost(e) { try { var data = JSON.parse(e.postData.contents); if (data.action === 'simpan') { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_PENGAWAS_NAME); var tz = ss.getSpreadsheetTimeZone(); var timestamp = Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd HH:mm:ss'); sheet.appendRow([timestamp, data.ruang, data.pengawas]); return jsonOutput({ status: 'ok' }); } if (data.action === 'otorisasiManual') { // Konfirmasi manual untuk siswa yang kamera HP-nya gagal scan // QR Token. Ditulis ke sheet "Otorisasi" di spreadsheet SUMBER // TOKEN (#2) — BUKAN di spreadsheet PENGAWAS (#1) — supaya // asesmen.html bisa membacanya langsung lewat gviz tanpa perlu // panggil Apps Script apa pun. var nisn = String(data.nisn || '').trim(); if (!nisn) return jsonOutput({ status: 'error', message: 'NISN kosong' }); var ssToken = SpreadsheetApp.openById(SHEET_TOKEN_ID); var sheetOt = ssToken.getSheetByName(SHEET_OTORISASI_NAME); if (!sheetOt) { sheetOt = ssToken.insertSheet(SHEET_OTORISASI_NAME); sheetOt.appendRow(['NISN', 'Timestamp', 'Ruang', 'Pengawas']); } var tzTok = ssToken.getSpreadsheetTimeZone(); var waktu = Utilities.formatDate(new Date(), tzTok, "yyyy-MM-dd'T'HH:mm:ss"); // PENTING: kolom NISN (A) DIPAKSA format Teks ('@') SEBELUM diisi. // Kalau tidak, NISN yang angka murni akan otomatis dikonversi // Google Sheets jadi Number — bikin query gviz di asesmen.html // (yang membandingkan sebagai teks berkutip) tidak pernah cocok, // sehingga status konfirmasi selalu dianggap "belum" walau sudah // dikonfirmasi pengawas. var barisBaru = sheetOt.getLastRow() + 1; sheetOt.getRange(barisBaru, 1).setNumberFormat('@').setValue(nisn); sheetOt.getRange(barisBaru, 2, 1, 3).setValues([[waktu, data.ruang || '', data.pengawas || '']]); return jsonOutput({ status: 'ok', nisn: nisn, waktu: waktu }); } return jsonOutput({ status: 'error', message: 'Aksi tidak dikenali' }); } catch (err) { return jsonOutput({ status: 'error', message: err.message }); } } function doGet(e) { try { var action = e.parameter.action; if (action === 'token') { var ssToken = SpreadsheetApp.openById(SHEET_TOKEN_ID); var sheet = ssToken.getSheetByName(SHEET_TOKEN_NAME); var value = sheet.getRange(SHEET_TOKEN_CELL).getDisplayValue(); return jsonOutput({ status: 'ok', value: String(value), updated: new Date().toISOString() }); } if (action === 'guru') { var ssGuru = SpreadsheetApp.getActiveSpreadsheet(); var sheetGuru = ssGuru.getSheetByName(SHEET_GURU_NAME); var lastRow = sheetGuru.getLastRow(); var daftar = []; if (lastRow >= 1) { var range = sheetGuru.getRange(SHEET_GURU_COLUMN + '1:' + SHEET_GURU_COLUMN + lastRow); var values = range.getValues(); var headerKeywords = ['nama', 'nama guru', 'nama pengawas', 'guru', 'pengawas']; values.forEach(function(row, idx){ var val = String(row[0]).trim(); if (!val) return; var isHeader = (idx === 0) && headerKeywords.indexOf(val.toLowerCase()) !== -1; if (!isHeader) daftar.push(val); }); } return jsonOutput({ status: 'ok', daftar: daftar }); } return jsonOutput({ status: 'error', message: 'Aksi tidak dikenali' }); } catch (err) { return jsonOutput({ status: 'error', message: err.message }); } } function jsonOutput(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // ================================================================ // EXPORT DAFTAR GURU KE GITHUB — jalankan MANUAL (pilih fungsi ini // di dropdown, klik Run) setiap kali ada guru baru ditambahkan atau // nama diubah di Sheet2!A:A. Datanya jarang berubah jadi TIDAK perlu // trigger otomatis — cukup dijalankan sesekali saat memang ada // perubahan, supaya pengawas-ruang.html tetap cepat (baca dari // GitHub, bukan Apps Script tiap kali dibuka). // // CATATAN: perlu GITHUB_TOKEN di Script Properties (Project Settings // → Script Properties → Add → GITHUB_TOKEN → token GitHub scope 'repo'). // ================================================================ var GITHUB_USER_GURU = 'SMPN1KALIWEDI'; var GITHUB_REPO_GURU = 'ujian-smpn1kaliwedi'; var GITHUB_BRANCH_GURU = 'main'; var GITHUB_PATH_GURU = 'data/guru.csv'; function exportGuruKeGithub() { var ssGuru = SpreadsheetApp.getActiveSpreadsheet(); var sheetGuru = ssGuru.getSheetByName(SHEET_GURU_NAME); var lastRow = sheetGuru.getLastRow(); var daftar = []; if (lastRow >= 1) { var range = sheetGuru.getRange(SHEET_GURU_COLUMN + '1:' + SHEET_GURU_COLUMN + lastRow); var values = range.getValues(); var headerKeywords = ['nama', 'nama guru', 'nama pengawas', 'guru', 'pengawas']; values.forEach(function(row, idx){ var val = String(row[0]).trim(); if (!val) return; var isHeader = (idx === 0) && headerKeywords.indexOf(val.toLowerCase()) !== -1; if (!isHeader) daftar.push(val); }); } var csvText = daftar.join('\n'); var token = PropertiesService.getScriptProperties().getProperty('GITHUB_TOKEN'); if (!token) throw new Error('GITHUB_TOKEN belum diatur di Script Properties.'); var url = 'https://api.github.com/repos/' + GITHUB_USER_GURU + '/' + GITHUB_REPO_GURU + '/contents/' + GITHUB_PATH_GURU; var encoded = Utilities.base64Encode(csvText, Utilities.Charset.UTF_8); var sha = null; var cek = UrlFetchApp.fetch(url, { method: 'GET', headers: { 'Authorization': 'token ' + token, 'Accept': 'application/vnd.github.v3+json' }, muteHttpExceptions: true, }); if (cek.getResponseCode() === 200) sha = JSON.parse(cek.getContentText()).sha; var payload = { message: 'Update guru.csv - ' + new Date().toISOString(), content: encoded, branch: GITHUB_BRANCH_GURU, }; if (sha) payload.sha = sha; var res = UrlFetchApp.fetch(url, { method: 'PUT', contentType: 'application/json', headers: { 'Authorization': 'token ' + token, 'Accept': 'application/vnd.github.v3+json' }, payload: JSON.stringify(payload), muteHttpExceptions: true, }); var kode = res.getResponseCode(); if (kode !== 200 && kode !== 201) { throw new Error('Gagal upload guru.csv. Kode: ' + kode + ' — ' + res.getContentText()); } console.log('✅ guru.csv berhasil di-export (' + daftar.length + ' nama)'); return { status: 'ok', jumlah: daftar.length }; }
